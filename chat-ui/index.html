    聊天页面的端口：chat UI 设计

    在本小节中，我们将介绍聊天页面的UI设计过程，通过多轮迭代逐步完善用户体验。至目前为止，我们一共有三轮认证： - 第一轮验证了浏览器能否成功向 FastAPI /chat 发送 JSON 请求，并把返回结果显示出来。 - 第二轮则围绕连续对话、请求状态和异常处理做一些基础的反馈，把单次的API调用做到更完善的、最轻量的聊天引用。 - 第三轮在RAG第一版本的开发完成之后，根据新增的功能增加了更多的视觉效果和互动优化。

    Date: 2026-09-12 15:59:31

    开发顺序：← 上一节：公网访问 ｜ 下一节：RAG项目总览 →

    1. 第一轮迭代

    我们的目的是让fastapi项目中写好的/chat功能在网页端上线。当时目标是先验证浏览器能否成功向 FastAPI /chat 发送 JSON 请求，并把返回结果显示出来。

    所以最初的目标是做出最简单的UI要求：一处能给用户输入文本的地方，一个发送文本的按钮，一处能显示聊天记录的聊天框。

    代码如下：

    # 聊天页面

<input id="message" type="text" placeholder="请输入消息">
<button onclick="sendMessage()">发送</button>

<p id="answer"></p>

<script>
    async function sendMessage(){
        const message = document.getElementById("message").value;
        const response = await fetch("https://infra01.xxxxxx.ts.net/chat", {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({ message:message })
        });
        const data = await response.json();
        document.getElementById("answer").innerText = data.answer;
    }

</script>


    我们能够发现，它本质上还是一个验证链路的原型。它每一次回答都会覆盖上一次内容。发送后输入框的内容也不会被清空。

    2. 第二轮迭代

    在第一轮的最简功能实现的基础上，我们进行进行更多的交互设计。这样，我们才不是单纯的一个输入框调用API，而是真的变成了一个最小可用的聊天应用。对原型的总结，我们需要增加的功能如下：

    1. 聊天记录按轮次显示，而不是永远覆盖answer
    2. 在代码主体区分 user / assistant 两种消息
    3. 发送后清空输入框
    4. 支持 Enter 发送
    5. 请求期间把按钮禁用或者显示“正在生成”或“发送中”
    6. 网络错误时显示错误信息
    7. 给聊天区域一个固定宽度和基本的气泡样式（提升更多的美术效果）

<script>
    function addMessage(role, text){
        const chatBox = document.getElementById("chat-box");

        const messageElement = document.createElement("div");
        messageElement.className = role;
        messageElement.innerText = text;

        chatBox.appendChild(messageElement);
        chatBox.scrollTop = chatBox.scrollHeight; // 自动滚动到底部
    }

    async function sendMessage(){
        const input = document.getElementById("message");
        const button = document.getElementById("send-button");

        const message = input.value.trim();//空格无效

        if(!message) return;

        addMessage("user", message);
        input.value = ""; //自动清空

        button.disabled = true; // 禁用按钮，防止重复点击
        button.innerText = "发送中..."; // 改变按钮文本

        try{
            const response = await fetch("https://infra01.xxxxxx.ts.net/chat", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({ message:message })
             });

            if(!response.ok){
                throw new Error("网络请求失败, HTTP error: "+response.status);
            }

            const data = await response.json();
            addMessage("assistant", data.answer);

        } catch (error){
            console.error("Error:", error);
            addMessage("assistant", "请求失败，请稍后再试。");
        } finally {
            button.disabled = false; // 恢复按钮
            button.innerText = "发送"; // 恢复按钮文本
            input.focus(); // 重新聚焦输入框
        }
    }

    document.getElementById("message").addEventListener("keypress", function(event){
        if(event.key === "Enter"){
            event.preventDefault(); // 阻止默认的回车行为
            sendMessage();
        }
    });

</script>

    我们能注意到，虽然第二轮迭代在界面上有了一些改进，但仍然存在一些问题，例如消息输入框和发送按钮的交互体验还可以进一步优化。某种意义上来说，顶部的聊天抬头有些太大了，占用了过多的屏幕空间。

    3. 第三轮迭代

    在RAG项目第一版本的开发完成之后，我们需要根据新功能做一些调整：

    1. 在assistant回复的时候，同时也能附上相关文档的链接气泡。回复的[**.md]在渲染后能够指向对应的文章页面。
    2. 根据模型API返回的内容，对其中的markdown格式文本进行渲染。
    3. 优化消息输入框和发送按钮的交互体验。

<script>
    function sourceToUrl(source) {
        if (source === "index.md") {
            return "/wiki-docs/";
        }
        if (source.endsWith(".md")) {
            return "/wiki-docs/" + source.replace(/\.md$/, "") + "/";
        }
        return "#";
    }

    function addMessage(role, text, sources = []){
        const chatBox = document.getElementById("chat-box");
        const messageElement = document.createElement("div");

        messageElement.className = role;

        if (role === "assistant"){
            messageElement.innerHTML = DOMPurify.sanitize(
                marked.parse(text)
            );

            if(sources.length > 0){
                const sourcesElement = document.createElement("div");
                sourcesElement.className = "sources";

                sources.forEach(source => {
                    const sourceLink = document.createElement("a");
                    sourceLink.className = "source-chip";
                    sourceLink.innerText = source;
                    sourceLink.href = sourceToUrl(source);
                    sourceLink.target = "_blank";

                    sourcesElement.appendChild(sourceLink);
                });
                messageElement.appendChild(sourcesElement);
            }

        }else{
                messageElement.innerText = text;
         }

        chatBox.appendChild(messageElement);
            chatBox.scrollTop = chatBox.scrollHeight; // 自动滚动到底部
    }

    async function sendMessage(){
        const input = document.getElementById("message");
        const button = document.getElementById("send-button");

        const message = input.value.trim();//空格无效

        if(!message) return;

        addMessage("user", message);
        input.value = ""; //自动清空

        button.disabled = true; // 禁用按钮，防止重复点击
        button.innerText = "发送中..."; // 改变按钮文本

        try{
            const response = await fetch("https://infra01.xxxxxx.ts.net/chat", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({ message:message })
             });

            if(!response.ok){
                throw new Error("网络请求失败, HTTP error: "+response.status);
            }

            const data = await response.json();
            addMessage("assistant", data.answer, data.sources || []);

        } catch (error){
            console.error("Error:", error);
            addMessage("assistant", "请求失败，请稍后再试。");
        } finally {
            button.disabled = false; // 恢复按钮
            button.innerText = "发送"; // 恢复按钮文本
            input.focus(); // 重新聚焦输入框
        }
    }

    document.getElementById("message").addEventListener("keydown", function(event){
        if(event.key === "Enter"){
            event.preventDefault(); // 阻止默认的回车行为
            sendMessage();
        }
    });

</script>
